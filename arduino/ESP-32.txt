#include <Wire.h>
#include <Adafruit_BNO08x.h>
#include <BLEDevice.h>
#include <BLEServer.h>
#include <BLEUtils.h>
#include <BLE2902.h>

// ---------- Pin assignments ----------
#define EMG1_PIN 34   // ADC1_CH6
#define EMG2_PIN 35   // ADC1_CH7

// I2C stays on the default pins
#define I2C_SDA 21
#define I2C_SCL 22

// ---------- IMUs: two fixed addresses, no multiplexer ----------
#define NUM_IMUS 2
const uint8_t imuAddr[NUM_IMUS] = {0x4A, 0x4B}; 

// ---------- Sampling config ----------
#define SAMPLE_RATE_HZ 500          
#define EMG_OVERSAMPLE 8            
#define IMU_REFRESH_EVERY_N 5       

// ---------- BLE Configuration ----------
// Standard Nordic UART Service (NUS) UUIDs (recognized by terminal apps)
#define SERVICE_UUID           "6E400001-B5A3-F393-E0A9-E50E24DCCA9E"
#define TX_CHARACTERISTIC_UUID "6E400003-B5A3-F393-E0A9-E50E24DCCA9E"

BLEServer *pServer = NULL;
BLECharacteristic *pTxCharacteristic = NULL;
bool deviceConnected = false;

// Server callback class to track connection status
class MyServerCallbacks: public BLEServerCallbacks {
    void onConnect(BLEServer* pServer) {
      deviceConnected = true;
    };
    void onDisconnect(BLEServer* pServer) {
      deviceConnected = false;
      // Restart advertising immediately so it can be refound
      pServer->getAdvertising()->start();
    }
};

hw_timer_t *timer = NULL;
volatile bool sampleFlag = false;

void IRAM_ATTR onTimer() {
  sampleFlag = true;
}

// ---------- IMU objects/state ----------
Adafruit_BNO08x bno[NUM_IMUS];
bool imuConnected[NUM_IMUS] = {false, false};

struct ImuReading {
  bool valid;
  float ax, ay, az;
  float gx, gy, gz;
};
ImuReading lastImu[NUM_IMUS];

void setReports(Adafruit_BNO08x &sensor) {
  sensor.enableReport(SH2_ACCELEROMETER);
  sensor.enableReport(SH2_GYROSCOPE_CALIBRATED);
}

void setup() {
  Serial.begin(115200);
  
  // --- BLE Initialisation ---
  BLEDevice::init("ESP32_EMG_IMU"); // Broadcast name
  
  pServer = BLEDevice::createServer();
  pServer->setCallbacks(new MyServerCallbacks());

  // Create the UART service
  BLEService *pService = pServer->createService(SERVICE_UUID);

  // Create the TX Characteristic for streaming data OUT
  pTxCharacteristic = pService->createCharacteristic(
                        TX_CHARACTERISTIC_UUID,
                        BLECharacteristic::PROPERTY_NOTIFY
                      );
                      
  // Required for client apps (like phone terminals) to subscribe to notifications
  pTxCharacteristic->addDescriptor(new BLE2902());

  pService->start();

  // Configure and start advertising
  BLEAdvertising *pAdvertising = BLEDevice::getAdvertising();
  pAdvertising->addServiceUUID(SERVICE_UUID);
  pAdvertising->setScanResponse(true);
  pAdvertising->setMinPreferred(0x06);  // helps with iPhone connection speed
  pAdvertising->setMaxPreferred(0x12);
  BLEDevice::startAdvertising();
  Serial.println("BLE UART ready. Awaiting connection...");

  // --- ADC setup ---
  analogReadResolution(12);                    
  analogSetPinAttenuation(EMG1_PIN, ADC_11db);  
  analogSetPinAttenuation(EMG2_PIN, ADC_11db);

  // --- I2C + IMUs ---
  Wire.begin(I2C_SDA, I2C_SCL);
  Wire.setClock(400000); 

  for (int i = 0; i < NUM_IMUS; i++) {
    if (bno[i].begin_I2C(imuAddr[i])) {
      setReports(bno[i]);
      imuConnected[i] = true;
      Serial.print("{\"status\":\"IMU ");
      Serial.print(i);
      Serial.print(" connected at 0x");
      Serial.print(imuAddr[i], HEX);
      Serial.println("\"}");
    } else {
      imuConnected[i] = false;
      Serial.print("{\"status\":\"IMU ");
      Serial.print(i);
      Serial.print(" NOT FOUND at 0x");
      Serial.print(imuAddr[i], HEX);
      Serial.println("\"}");
    }
    lastImu[i] = {false, 0, 0, 0, 0, 0, 0};
  }

  // --- Hardware timer ---
  timer = timerBegin(1000000);              
  timerAttachInterrupt(timer, &onTimer);
  uint64_t periodUs = 1000000UL / SAMPLE_RATE_HZ;
  timerAlarm(timer, periodUs, true, 0);     
}

int readEmgAveraged(int pin) {
  long sum = 0;
  for (int i = 0; i < EMG_OVERSAMPLE; i++) {
    sum += analogRead(pin);
  }
  return sum / EMG_OVERSAMPLE;
}

void refreshImu(int index) {
  if (!imuConnected[index]) return;

  sh2_SensorValue_t event;
  unsigned long start = millis();
  while (millis() - start < 3) { 
    if (bno[index].getSensorEvent(&event)) {
      if (event.sensorId == SH2_ACCELEROMETER) {
        lastImu[index].ax = event.un.accelerometer.x;
        lastImu[index].ay = event.un.accelerometer.y;
        lastImu[index].az = event.un.accelerometer.z;
        lastImu[index].valid = true;
      } else if (event.sensorId == SH2_GYROSCOPE_CALIBRATED) {
        lastImu[index].gx = event.un.gyroscope.x;
        lastImu[index].gy = event.un.gyroscope.y;
        lastImu[index].gz = event.un.gyroscope.z;
        lastImu[index].valid = true;
      }
    }
  }
}

// Global buffer to construct the string for BLE
String bleBuffer = "";

void printPacket(int emg1, int emg2) {
  // Clear the buffer string
  bleBuffer = "";
  
  // Format local Serial and build BLE string concurrently
  Serial.print("{\"emg1\":"); Serial.print(emg1);
  bleBuffer += "{\"emg1\":" + String(emg1);

  Serial.print(",\"emg2\":"); Serial.print(emg2);
  bleBuffer += ",\"emg2\":" + String(emg2);

  Serial.print(",\"imus\":[");
  bleBuffer += ",\"imus\":[";

  for (int i = 0; i < NUM_IMUS; i++) {
    if (i > 0) {
      Serial.print(",");
      bleBuffer += ",";
    }
    if (lastImu[i].valid) {
      // Build IMU node (Truncating decimals slightly saves massive payload size)
      String imuStr = "{\"ax\":" + String(lastImu[i].ax, 2) +
                      ",\"ay\":" + String(lastImu[i].ay, 2) +
                      ",\"az\":" + String(lastImu[i].az, 2) +
                      ",\"gx\":" + String(lastImu[i].gx, 2) +
                      ",\"gy\":" + String(lastImu[i].gy, 2) +
                      ",\"gz\":" + String(lastImu[i].gz, 2) + "}";
                      
      Serial.print(imuStr);
      bleBuffer += imuStr;
    } else {
      Serial.print("null");
      bleBuffer += "null";
    }
  }
  Serial.println("]}");
  bleBuffer += "]}\n";

  // Send over BLE if a client is connected
  if (deviceConnected) {
    pTxCharacteristic->setValue(bleBuffer.c_str());
    pTxCharacteristic->notify();
  }
}

void loop() {
  static uint32_t tickCount = 0;

  if (sampleFlag) {
    sampleFlag = false;
    tickCount++;

    int emg1 = readEmgAveraged(EMG1_PIN);
    int emg2 = readEmgAveraged(EMG2_PIN);

    if (tickCount % IMU_REFRESH_EVERY_N == 0) {
      for (int i = 0; i < NUM_IMUS; i++) {
        refreshImu(i);
      }
    }

    printPacket(emg1, emg2);
  }
}
