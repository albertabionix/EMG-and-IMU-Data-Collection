#include <Wire.h>
#include <Adafruit_LSM6DS.h>
#include <Adafruit_ISM330DHCX.h> 
#include <BluetoothSerial.h>

#define EMG1_PIN 34 
#define EMG2_PIN 35 
#define I2C_SDA 21
#define I2C_SCL 22 

#define NUM_IMUS 2
const uint8_t imuAddr[NUM_IMUS] = {0x6A, 0x6B}; 

// --- 1000Hz Target Configuration ---
#define SAMPLE_RATE_HZ 1000
#define IMU_REFRESH_EVERY_N 10 // Reads IMUs at 100Hz (Every 10ms) to preserve ADC bandwidth
#define CLASSIC_SEND_EVERY_N 1 // Send every acquired sample: 1000 Bluetooth Classic packets per second

// Compact Binary Structure (Total: 28 bytes)
struct __attribute__((packed)) SensorPacket {
  uint16_t emg1;
  uint16_t emg2;
  struct __attribute__((packed)) {
    int16_t ax, ay, az;
    int16_t gx, gy, gz;
  } imu[NUM_IMUS];
};

BluetoothSerial SerialBT;

QueueHandle_t dataQueue;
const int QUEUE_SIZE = 200; 

hw_timer_t *timer = NULL;
SemaphoreHandle_t timerSemaphore;

void IRAM_ATTR onTimer() {
  BaseType_t xHigherPriorityTaskWoken = pdFALSE;
  xSemaphoreGiveFromISR(timerSemaphore, &xHigherPriorityTaskWoken);
  if (xHigherPriorityTaskWoken) {
    portYIELD_FROM_ISR();
  }
}

Adafruit_ISM330DHCX ism[NUM_IMUS]; 
bool imuConnected[NUM_IMUS] = {false, false};

void DataAcquisitionTask(void * pvParameters);
void BleTransmissionTask(void * pvParameters);

void setup() {
  // Clean high-speed serial
  Serial.begin(921600); 
  delay(500); // Give the serial monitor time to connect
  Serial.println("Booting ESP32...");

  timerSemaphore = xSemaphoreCreateBinary();
  dataQueue = xQueueCreate(QUEUE_SIZE, sizeof(SensorPacket));

  // --- Bluetooth Classic SPP setup ---
  if (!SerialBT.begin("ESP32_EMG_IMU_1KHZ")) {
    Serial.println("Bluetooth Classic initialization failed.");
  } else {
    Serial.println("Bluetooth Classic ready. Pair with ESP32_EMG_IMU_1KHZ.");
  }

  // --- High-Speed ADC Optimization ---
  analogReadResolution(12);
  analogSetPinAttenuation(EMG1_PIN, ADC_11db);
  analogSetPinAttenuation(EMG2_PIN, ADC_11db);

  // --- Reliable I2C Setup ---
  Wire.begin(I2C_SDA, I2C_SCL);
  Wire.setClock(400000); // Safe, rock-solid 400kHz Fast Mode

  for (int i = 0; i < NUM_IMUS; i++) {
    if (ism[i].begin_I2C(imuAddr[i])) { 
      ism[i].setAccelRange(LSM6DS_ACCEL_RANGE_4_G);
      ism[i].setGyroRange(LSM6DS_GYRO_RANGE_500_DPS);
      imuConnected[i] = true;
      Serial.print("IMU "); Serial.print(i); Serial.println(" initialized successfully.");
    } else {
      Serial.print("IMU "); Serial.print(i); Serial.println(" NOT found!");
    }
  }

  // Core 1 handles strict 1ms hardware polling loop
  xTaskCreatePinnedToCore(DataAcquisitionTask, "AcqTask", 4096, NULL, 3, NULL, 1); 
  // Core 0 processes background BLE and Serial output streams
  xTaskCreatePinnedToCore(BleTransmissionTask, "BleTask", 4096, NULL, 1, NULL, 0); 

  // --- Hardware Timer Configuration (1000Hz) ---
  timer = timerBegin(1000000); 
  timerAttachInterrupt(timer, &onTimer);
  uint64_t periodUs = 1000000UL / SAMPLE_RATE_HZ; // 1000us = 1ms
  timerAlarm(timer, periodUs, true, 0);
  
  Serial.println("System fully ready!");
}

void DataAcquisitionTask(void * pvParameters) {
  uint32_t tickCount = 0;
  SensorPacket currentPacket = {0};
  
  // Create real local event containers instead of passing NULL pointers
  sensors_event_t dummyAccel, dummyGyro, dummyTemp;

  while(1) {
    if (xSemaphoreTake(timerSemaphore, portMAX_DELAY) == pdTRUE) {
      tickCount++;

      // Read analog EMG signals immediately (Takes ~10 microseconds)
      currentPacket.emg1 = analogRead(EMG1_PIN);
      currentPacket.emg2 = analogRead(EMG2_PIN);

      // Decouple IMU reads to every 10th loop tick (100Hz real data)
      if (tickCount % IMU_REFRESH_EVERY_N == 0) {
        for (int i = 0; i < NUM_IMUS; i++) {
          if (imuConnected[i]) {
            
            // Pass real tracking references to force the internal I2C buffer to fill
            ism[i].getEvent(&dummyAccel, &dummyGyro, &dummyTemp); 

            // Extract the freshly cached raw int16_t variables directly 
            currentPacket.imu[i].ax = ism[i].rawAccX;
            currentPacket.imu[i].ay = ism[i].rawAccY;
            currentPacket.imu[i].az = ism[i].rawAccZ;
            
            currentPacket.imu[i].gx = ism[i].rawGyroX;
            currentPacket.imu[i].gy = ism[i].rawGyroY;
            currentPacket.imu[i].gz = ism[i].rawGyroZ;
          }
        }
      }

      // Offload payload array to data queue (non-blocking)
      xQueueSend(dataQueue, &currentPacket, 0);
    }
  }
}


void BleTransmissionTask(void * pvParameters) {
  SensorPacket txPacket;
  uint32_t classicTickCount = 0;

  while(1) {
    if (xQueueReceive(dataQueue, &txPacket, portMAX_DELAY) == pdTRUE) {
      
      // Fast raw CSV stream layout for Serial Plotter evaluation
      Serial.print(txPacket.emg1); Serial.print(",");
      Serial.println(txPacket.emg2);

      // Ship compact 28-byte binary payload over Bluetooth Classic SPP
      classicTickCount++;
      if (SerialBT.hasClient() && classicTickCount % CLASSIC_SEND_EVERY_N == 0) {
        SerialBT.write((const uint8_t*)&txPacket, sizeof(SensorPacket));
      }
    }
  }
}

void loop() {
  vTaskDelete(NULL); 
}
